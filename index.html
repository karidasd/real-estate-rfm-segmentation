🗄️ Η λογική σε SQL (`WHERE direction = 'inbound'`)

WITH inbound_activity AS (
  SELECT lead_id, MAX(event_date) AS last_inbound_date,
         COUNT(*) AS signals_count,
         MAX(CASE WHEN event_type='property_visit' THEN 1 ELSE 0 END) AS has_visit
  FROM sample_interactions
  WHERE direction = 'inbound' -- Αγνοούμε δικά μας follow-ups!
  GROUP BY lead_id
)
SELECT l.lead_id, l.full_name, l.budget,
  CASE
    WHEN days_since_inbound <= 60 AND (
      has_visit = 1 OR signals_count >= 4
      OR (signals_count BETWEEN 2 AND 3 AND budget >= 200000)
    ) THEN 'Έτοιμος αγοραστής'
    WHEN days_since_inbound <= 60  THEN 'Νέος ενδιαφερόμενος'
    WHEN days_since_inbound <= 180 THEN 'Σε σκέψη'
    WHEN days_since_inbound <= 365 THEN 'Σε ψύξη'
    ELSE 'Σε νάρκη'
  END AS segment
FROM sample_crm_leads l
LEFT JOIN inbound_activity a ON l.lead_id = a.lead_id;
